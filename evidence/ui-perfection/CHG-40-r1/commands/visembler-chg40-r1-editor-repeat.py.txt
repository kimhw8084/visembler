def pct(values,p):
    if not values:return None
    values=sorted(values); idx=(len(values)-1)*p/100; lo=int(idx); hi=min(lo+1,len(values)-1); frac=idx-lo
    return values[lo]+(values[hi]-values[lo])*frac
