def audit_page(page, state_id, viewport, theme, motion, model_kind, events, method=None):

  try:
   if page.locator('#libraryToggle').get_attribute('aria-pressed')!='true': page.locator('#libraryToggle').click(); page.wait_for_timeout(120)
  except Exception: pass
